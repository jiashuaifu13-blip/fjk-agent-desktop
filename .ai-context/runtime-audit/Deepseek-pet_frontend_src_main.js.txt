 * Message rendering
 * ==========================

function renderMessages(
  messages
) {

  messageList.innerHTML =
    ''


  if (
    messages.length === 0
  ) {

    const empty =
      document.createElement(
        'div'
      )

    empty.className =
      'empty-message'

    empty.textContent =
      '鍜屾瀹犺鐐逛粈涔堝惂銆?


    messageList.appendChild(
      empty
    )

    return
  }


  for (
    const message
    of messages
  ) {

    const item =
      document.createElement(
        'div'
      )


    item.className =
      `message message-${message.role}`


    const role =
      document.createElement(
        'div'
      )


    role.className =
      'message-role'


    role.textContent =
      message.role === 'user'
        ? 'You'
        : 'Pet'


    const content =
      document.createElement(
        'div'
      )


    content.className =
      'message-content'


    content.textContent =
      message.content ||
      '鈥?


    item.append(
      role,
      content
    )


    messageList.appendChild(
      item
    )
  }


  messageList.scrollTop =
    messageList.scrollHeight
}


/*
 * ==========================
 * Floating Chat Controller
 * ==========================
 */

function setupFloatingChat() {

  let collapsed =
    false

  let dragging =
    false

  let pointerId =
    null

  let startPointerX =
    0

  let startPointerY =
    0

  let startLeft =
    0

  let startTop =
    0


  function clampPosition(
    left,
    top
  ) {

    const rect =
      chatScene
        .getBoundingClientRect()


    const padding =
      10


    const maxLeft =
      Math.max(
        padding,
        window.innerWidth -
        rect.width -
        padding
      )


    const maxTop =
      Math.max(
        padding,
        window.innerHeight -
        rect.height -
        padding
      )


    return {
      left:
        Math.min(
          Math.max(
            padding,
            left
          ),
          maxLeft
        ),

      top:
        Math.min(
          Math.max(
            padding,
            top
          ),
          maxTop
        ),
    }
  }


  function applyPosition(
    left,
    top
  ) {

    const position =
      clampPosition(
        left,
        top
      )


    chatScene.style.left =
      `${position.left}px`

    chatScene.style.top =
      `${position.top}px`

    chatScene.style.right =
      'auto'

    chatScene.style.bottom =
      'auto'


    return position
  }


  function savePosition() {

    const rect =
      chatScene
        .getBoundingClientRect()


    try {

      localStorage.setItem(
        CHAT_POSITION_KEY,
        JSON.stringify({
          left:
            rect.left,

          top:
            rect.top,
        })
      )

    } catch {
      /*
       * localStorage 涓嶅彲鐢ㄦ椂
       * 鐩存帴蹇界暐銆?       */
    }
  }


  function restorePosition() {

    try {

      const raw =
        localStorage.getItem(
          CHAT_POSITION_KEY
        )


      if (raw) {

        const saved =
          JSON.parse(
            raw
          )


        if (
          Number.isFinite(
            saved.left
          ) &&
          Number.isFinite(
            saved.top
          )
        ) {

          applyPosition(
            saved.left,
            saved.top
          )

          return
        }
      }

    } catch {
      /*
       * 鏁版嵁鎹熷潖鏃跺洖榛樿浣嶇疆銆?       */
    }


    /*
     * 榛樿浠嶆帴杩戝弬鑰冨浘锛?     * 妗屽疇宸︿晶銆佸睆骞曞亸鍙充笅銆?     *
     * 鍙槸鍒濆浣嶇疆锛?     * 涔嬪悗鐢ㄦ埛鍙互闅忎究鎷栥€?     */
    const rect =
      chatScene
        .getBoundingClientRect()


    const defaultLeft =
      window.innerWidth -
      rect.width -
      300


    const defaultTop =
      window.innerHeight -
      rect.height -
      54


    applyPosition(
      defaultLeft,
      defaultTop
    )
  }


  function normalizePosition() {

    const rect =
      chatScene
        .getBoundingClientRect()


    applyPosition(
      rect.left,
      rect.top
    )


    savePosition()
  }


  function setCollapsed(
    value
  ) {

    collapsed =
      value


    chatShell.classList.toggle(
      'is-collapsed',
      collapsed
    )


    chatContent.hidden =
      collapsed


    chatDragHandle.hidden =
      collapsed


    chatCollapsed.hidden =
      !collapsed


    /*
     * 鎶樺彔鍚庡昂瀵稿彉鍖栵紝
     * 涓嬩竴甯ч噸鏂板仛灞忓箷杈圭晫妫€鏌ャ€?     */
    requestAnimationFrame(
      normalizePosition
    )
  }


  function startDrag(
    event,
    handle
  ) {

    if (
      event.button !== undefined &&
      event.button !== 0
    ) {
      return
    }


    event.preventDefault()


    const rect =
      chatScene
        .getBoundingClientRect()


    dragging =
      true

    pointerId =
      event.pointerId


    startPointerX =
      event.clientX

    startPointerY =
      event.clientY


    startLeft =
      rect.left

    startTop =
      rect.top


    chatScene.classList.add(
      'is-dragging'
    )


    handle.setPointerCapture(
      event.pointerId
    )
  }


  function moveDrag(
    event
  ) {

    if (
      !dragging ||
      event.pointerId !==
      pointerId
    ) {
      return
    }


    const deltaX =
      event.clientX -
      startPointerX


    const deltaY =
      event.clientY -
      startPointerY


    applyPosition(
      startLeft + deltaX,
      startTop + deltaY
    )
  }


  function endDrag(
    event,
    handle
  ) {

    if (
      !dragging ||
      event.pointerId !==
      pointerId
    ) {
      return
    }


    try {

      if (
        handle.hasPointerCapture(
          event.pointerId
        )
      ) {

        handle.releasePointerCapture(
          event.pointerId
        )
      }

    } catch {
      // ignore
    }


    dragging =
      false

    pointerId =
      null


    chatScene.classList.remove(
      'is-dragging'
    )


    savePosition()
  }


  /*
   * 灞曞紑鐘舵€侊細
   * 椤堕儴缁嗘潯鎷栧姩銆?   */

  chatDragHandle.addEventListener(
    'pointerdown',
    (event) => {
      startDrag(
        event,
        chatDragHandle
      )
    }
  )


  chatDragHandle.addEventListener(
    'pointermove',
    moveDrag
  )


  chatDragHandle.addEventListener(
    'pointerup',
    (event) => {
      endDrag(
        event,
        chatDragHandle
      )
    }
  )


  chatDragHandle.addEventListener(
    'pointercancel',
    (event) => {
      endDrag(
        event,
        chatDragHandle
      )
    }
  )


  /*
   * 鎶樺彔鐘舵€侊細
   * 鏁翠釜鑳跺泭鏃㈣兘鎷栵紝涔熻兘鐐瑰嚮灞曞紑銆?   */

  let collapsedMoved =
    false

  let collapsedStartX =
    0

  let collapsedStartY =
    0


  chatCollapsed.addEventListener(
    'pointerdown',
    (event) => {

      collapsedMoved =
        false

      collapsedStartX =
        event.clientX

      collapsedStartY =
        event.clientY


      startDrag(
        event,
        chatCollapsed
      )
    }
  )


  chatCollapsed.addEventListener(
    'pointermove',
    (event) => {

      if (
        dragging
      ) {

        const distance =
          Math.hypot(
            event.clientX -
              collapsedStartX,

            event.clientY -
              collapsedStartY
          )


        if (
          distance >
          5
        ) {
          collapsedMoved =
            true
        }
      }


      moveDrag(
        event
      )
    }
  )


  chatCollapsed.addEventListener(
    'pointerup',
    (event) => {

      endDrag(
        event,
        chatCollapsed
      )


      if (
        !collapsedMoved
      ) {
        setCollapsed(
          false
        )
      }
    }
  )


  chatCollapsed.addEventListener(
    'pointercancel',
    (event) => {

      endDrag(
        event,
        chatCollapsed
      )
    }
  )


  toggleChatButton.addEventListener(
    'click',
    () => {

      setCollapsed(
        true
      )
    }
  )


  window.addEventListener(
    'resize',
    normalizePosition
  )


  /*
   * 绛夋祻瑙堝櫒瀹屾垚甯冨眬鍚?   * 鍐嶆仮澶嶄綅缃€?   */
  requestAnimationFrame(
    restorePosition
  )


  return {
    setCollapsed,

    get collapsed() {
      return collapsed
    },
  }
}


/*
 * ==========================
 * Main
 * ==========================

async function main() {

  try {

    setupDebug()


    /*
     * ==========================
     * dsh1
     * ==========================
     */

    const loader =
      new PetDataLoader()


    const {
      actions,
    } =
      await loader.load()


    const library =
      new AnimationLibrary(
        actions
      )


    const cache =
      new AnimationCache()


    const renderer =
      new PetRenderer(
        canvas
      )


    const audioManager =
      new PetAudioManager()


    const animator =
      new PetAnimator({
        renderer,
        library,
        cache,
        audioManager,
      })


    const stateMachine =
      new PetStateMachine({
        animator,
      })


    animator.onAnimationChange =
      (
        action,
        info
      ) => {

        actionNameElement.textContent =
          action.id


        fpsElement.textContent =
          action
            .spritesheet
            .fps


        frameCountElement.textContent =
          action
            .spritesheet
            .frameCount


        cacheCountElement.textContent =
          cache.size


        audioStateElement.textContent =
          info.hasAudio
            ? '鏈?
            : '鏃?


        clockStateElement.textContent =
          info.usingAudioClock
            ? 'Audio'
            : 'Performance'
      }


    animator.onError =
      (error) => {

        console.error(
          '[Animator Error]',
          error
        )


        statusElement.textContent =
          `鍔ㄧ敾閿欒锛?{error.message}`
      }


    stateMachine.onStateChange =
      ({
        from,
        to,
        reason,
      }) => {

        stateNameElement.textContent =
          to


        previousStateElement.textContent =
          from ?? '-'


        transitionReasonElement.textContent =
          reason


        cacheCountElement.textContent =
          cache.size
      }


    stateMachine.onStateError =
      (error) => {

        console.error(
          '[State Error]',
          error
        )
      }


    await stateMachine.setState(
      PetStates.IDLE,
      {
        reason:
          'application-ready',
      }
    )


    /*
     * ==========================
     * Pet interaction
     * ==========================
     */

    const hitTest =
      new PetHitTest({
        canvas,

        alphaThreshold:
          25,
      })


    const interactionController =
      new PetInteractionController({
        root:
          petRoot,

        canvas,

        stateMachine,

        hitTest,

        dragThreshold:
          6,
      })


    interactionController
      .onInteraction =
      (event) => {

        interactionStateElement
          .textContent =
          event.type
      }


    interactionController.mount()


    /*
     * ==========================
     * Floating Chat
     * ==========================
     */

    const floatingChat =
      setupFloatingChat()


    /*
     * ==========================
     * Mock Chat
     * ==========================
     */

    const mockChatClient =
      new MockChatClient({
        thinkingDelay:
          900,

        tokenDelay:
          45,
      })


    const chatController =
      new ChatController({
        client:
          mockChatClient,

        stateMachine,
      })


    chatController
      .onMessageChange =
      (messages) => {

        renderMessages(
          messages
        )
      }


    chatController
      .onGeneratingChange =
      (generating) => {

        sendButton.disabled =
          generating


        stopButton.disabled =
          !generating


        chatInput.disabled =
          generating


        chatStatusElement.textContent =
          generating
            ? '姝ｅ湪鍥炲鈥?
            : 'Mock 鍦ㄧ嚎'


        collapsedText.textContent =
          generating
            ? '姝ｅ湪鍥炲鈥?
            : '鍜屾垜鑱婂ぉ'
      }


    chatController
      .onBubbleChange =
      (content) => {

        const text =
          content.trim()


        if (!text) {

          petBubble.hidden =
            true

          petBubble.textContent =
            ''

          return
        }


        petBubble.hidden =
          false


        petBubble.textContent =
          text


        petBubble.scrollTop =
          petBubble.scrollHeight
      }


    chatController
      .onError =
      (error) => {

        console.error(
          '[Chat Error]',
          error
        )


        chatStatusElement.textContent =
          '閿欒'
      }


    chatForm.addEventListener(
      'submit',

      async (event) => {

        event.preventDefault()


        const content =
          chatInput.value.trim()


        if (!content) {
          return
        }


        chatInput.value =
          ''


        try {

          await chatController.send(
            content
          )

        } catch (error) {

          console.error(
            error
          )
        }
      }
    )


    chatInput.addEventListener(
      'keydown',

      (event) => {

        if (
          event.key ===
            'Enter' &&
          !event.shiftKey
        ) {

          event.preventDefault()


          if (
            !chatController
              .isGenerating
          ) {

            chatForm
              .requestSubmit()
          }
        }
      }
    )


    stopButton.addEventListener(
      'click',

      () => {

        chatController.stop()
      }
    )


    clearButton.addEventListener(
      'click',

      () => {

        chatController.clear()
      }
    )


    statusElement.textContent =
      'Phase 6锛欶loating Companion UI 宸插惎鍔?


    window.deepseekPet = {
      library,
      cache,
      renderer,
      animator,
      stateMachine,
      hitTest,
      interactionController,
      mockChatClient,
      chatController,
      floatingChat,
    }

  } catch (error) {

    console.error(
      'Deepseek Pet 鍒濆鍖栧け璐ワ細',
      error
    )


    statusElement.textContent =
      `鍒濆鍖栧け璐ワ細${error.message}`
  }
}


main()
