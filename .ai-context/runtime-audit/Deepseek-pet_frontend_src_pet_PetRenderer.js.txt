export class PetRenderer {
  constructor(canvas) {
    this.canvas = canvas
    this.ctx = canvas.getContext('2d')

    this.image = null
    this.action = null
  }

  setAnimation(
    action,
    image
  ) {
    this.action = action
    this.image = image

    const {
      frameWidth,
      frameHeight,
    } = action.spritesheet

    this.canvas.width =
      frameWidth

    this.canvas.height =
      frameHeight
  }

  clear() {
    this.ctx.clearRect(
      0,
      0,
      this.canvas.width,
      this.canvas.height
    )
  }

  renderFrame(frame) {
    if (
      !this.action ||
      !this.image
    ) {
      return
    }

    const {
      frameWidth,
      frameHeight,
      columns,
    } =
      this.action.spritesheet

    const column =
      frame % columns

    const row =
      Math.floor(
        frame / columns
      )

    const sourceX =
      column * frameWidth

    const sourceY =
      row * frameHeight

    this.clear()

    this.ctx.drawImage(
      this.image,

      sourceX,
      sourceY,

      frameWidth,
      frameHeight,

      0,
      0,

      this.canvas.width,
      this.canvas.height
    )
  }
}
