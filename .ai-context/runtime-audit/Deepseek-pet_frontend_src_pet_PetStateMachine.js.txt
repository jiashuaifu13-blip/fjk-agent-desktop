import {
  PetStates,
  PetStateConfig,
} from './PetStates.js'


export class PetStateMachine {

  constructor({
    animator,
    config = PetStateConfig,
  }) {

    this.animator =
      animator

    this.config =
      config

    this.currentState =
      PetStates.BOOT

    this.previousState =
      null

    this.transitionToken =
      0

    this.onStateChange =
      null

    this.onStateError =
      null


    /*
     * 鍔ㄧ敾鑷劧缁撴潫鏃讹紝
     * 鐘舵€佹満鍐冲畾涓嬩竴姝ュ幓鍝€?
     */

    this.animator.onAnimationEnd =
      async (action) => {

        const stateConfig =
          this.config[
            this.currentState
          ]

        if (
          !stateConfig ||
          !stateConfig.transient
        ) {
          return
        }

        const returnState =
          stateConfig.returnState ??
          PetStates.IDLE

        try {

          await this.setState(
            returnState,
            {
              reason:
                `animation-ended:${action.id}`,
            }
          )

        } catch (error) {

          this.handleError(
            error
          )
        }
      }
  }


  getStateConfig(state) {

    return (
      this.config[state] ??
      null
    )
  }


  canEnter(state) {

    if (
      state ===
      PetStates.BOOT
    ) {
      return false
    }

    return Boolean(
      this.getStateConfig(
        state
      )
    )
  }


  async setState(
    nextState,
    {
      force = false,
      reason = 'manual',
    } = {}
  ) {

    if (
      !this.canEnter(nextState)
    ) {
      throw new Error(
        `Unknown or unavailable pet state: ${nextState}`
      )
    }


    /*
     * 鐩稿悓鐘舵€侀粯璁や笉閲嶆柊鍚姩鍔ㄧ敾銆?
     */

    if (
      !force &&
      nextState ===
      this.currentState
    ) {
      return false
    }


    const token =
      ++this.transitionToken


    const previousState =
      this.currentState


    const stateConfig =
      this.getStateConfig(
        nextState
      )


    try {

      await this.animator.play(
        stateConfig.actionId,
        {
          loop:
            stateConfig.loop,
        }
      )


      /*
       * 濡傛灉 await 鏈熼棿鍙堝彂鐢熶簡鏂扮殑鐘舵€佸垏鎹紝
       * 褰撳墠杩欎竴杞粨鏋滀綔搴熴€?
       */

      if (
        token !==
        this.transitionToken
      ) {
        return false
      }


      this.previousState =
        previousState

      this.currentState =
        nextState


      if (
        typeof this.onStateChange ===
        'function'
      ) {

        this.onStateChange({
          from:
            previousState,

          to:
            nextState,

          reason,

          actionId:
            stateConfig.actionId,

          transient:
            Boolean(
              stateConfig.transient
            ),
        })
      }


      return true

    } catch (error) {

      this.handleError(
        error
      )

      throw error
    }
  }


  /*
   * 寮哄埗閲嶆柊鎾斁褰撳墠鐘舵€併€?
   */

  async restart() {

    if (
      this.currentState ===
      PetStates.BOOT
    ) {
      return false
    }

    return this.setState(
      this.currentState,
      {
        force: true,
        reason: 'restart',
      }
    )
  }


  /*
   * 鍥炲埌榛樿寰呮満鐘舵€併€?
   */

  async idle(
    reason = 'idle'
  ) {

    return this.setState(
      PetStates.IDLE,
      {
        reason,
      }
    )
  }


  handleError(error) {

    console.error(
      'PetStateMachine error:',
      error
    )


    if (
      typeof this.onStateError ===
      'function'
    ) {
      this.onStateError(
        error
      )
    }
  }

}
