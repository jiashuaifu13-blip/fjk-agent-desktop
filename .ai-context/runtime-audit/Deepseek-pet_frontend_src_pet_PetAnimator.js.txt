export class PetAnimator {
  constructor({
    renderer,
    library,
    cache,
    audioManager = null,
  }) {
    this.renderer = renderer
    this.library = library
    this.cache = cache

    this.audioManager =
      audioManager

    this.currentAction = null
    this.currentFrame = 0

    this.running = false
    this.looping = true

    this.startTime = 0
    this.animationFrameId = null

    this.usingAudioClock = false

    this.playToken = 0

    this.onAnimationChange = null
    this.onAnimationEnd = null
    this.onError = null
  }

  async play(
    actionId,
    {
      loop = true,
    } = {}
  ) {
    const token =
      ++this.playToken

    const action =
      this.library.get(actionId)

    if (!action) {
      throw new Error(
        `Unknown animation: ${actionId}`
      )
    }

    /*
     * 鍏堝姞杞界簿鐏靛浘銆?
     */

    const image =
      await this.cache.load(action)

    if (
      token !==
      this.playToken
    ) {
      return
    }

    /*
     * 鍋滄鍓嶄竴涓姩鐢诲拰闊抽銆?
     */

    this.stopAnimationLoop()

    if (this.audioManager) {
      await this.audioManager.stop()
    }

    if (
      token !==
      this.playToken
    ) {
      return
    }

    this.currentAction =
      action

    this.currentFrame =
      0

    this.looping =
      loop

    this.running =
      true

    this.usingAudioClock =
      false

    this.renderer.setAnimation(
      action,
      image
    )

    this.renderer.renderFrame(0)

    /*
     * 妫€鏌ュ姩浣滄槸鍚﹀叿鏈夐煶棰戙€?
     */

    if (
      this.audioManager &&
      this.audioManager.hasAudio(action)
    ) {
      try {
        const prepared =
          await this.audioManager.prepare(
            action
          )

        if (
          token !==
          this.playToken
        ) {
          return
        }

        if (prepared) {
          /*
           * 闊抽鍔ㄤ綔榛樿涓嶅惊鐜€?
           *
           * 鍥犱负闊抽鏈韩閫氬父鍙湁涓€娆°€?
           */
          this.looping = false

          await this.audioManager.play()

          this.usingAudioClock =
            true
        }
      } catch (error) {
        console.error(
          '闊抽鍔犺浇澶辫触锛屽洖閫€鍒版櫘閫氬姩鐢绘椂閽燂細',
          error
        )

        /*
         * 闊抽澶辫触涓嶈兘瀵艰嚧鍔ㄧ敾澶辫触銆?
         */

        this.usingAudioClock =
          false
      }
    }

    this.startTime =
      performance.now()

    if (
      typeof this.onAnimationChange ===
      'function'
    ) {
      this.onAnimationChange(
        action,
        {
          hasAudio:
            Boolean(
              this.audioManager?.hasAudio(
                action
              )
            ),

          usingAudioClock:
            this.usingAudioClock,
        }
      )
    }

    this.loop()
  }

  stopAnimationLoop() {
    this.running = false

    if (
      this.animationFrameId !==
      null
    ) {
      cancelAnimationFrame(
        this.animationFrameId
      )

      this.animationFrameId =
        null
    }
  }

  async stop() {
    ++this.playToken

    this.stopAnimationLoop()

    if (this.audioManager) {
      await this.audioManager.stop()
    }

    this.currentAction = null
    this.currentFrame = 0

    this.usingAudioClock =
      false
  }

  getElapsedSeconds() {
    /*
     * 鏈夐煶棰戯細
     *
     * 闊抽鏈韩灏辨槸鏃堕棿杞淬€?
     */

    if (
      this.usingAudioClock &&
      this.audioManager
    ) {
      return (
        this.audioManager.currentTime
      )
    }

    /*
     * 娌℃湁闊抽锛?
     *
     * 浣跨敤 performance.now()
     */

    return (
      performance.now() -
      this.startTime
    ) / 1000
  }

  loop = () => {
    if (
      !this.running ||
      !this.currentAction
    ) {
      return
    }

    try {
      const {
        fps,
        frameCount,
      } =
        this.currentAction
          .spritesheet

      const elapsedSeconds =
        this.getElapsedSeconds()

      let frame =
        Math.floor(
          elapsedSeconds * fps
        )

      /*
       * 闊抽鍔ㄤ綔锛?
       *
       * 濡傛灉闊抽缁撴潫锛?
       * 鍔ㄧ敾涔熺粨鏉熴€?
       */

      if (
        this.usingAudioClock &&
        this.audioManager?.ended
      ) {
        this.finishAnimation()
        return
      }

      if (this.looping) {
        frame =
          frame % frameCount
      } else {
        if (
          frame >=
          frameCount
        ) {
          this.finishAnimation()
          return
        }
      }

      if (
        frame !==
        this.currentFrame
      ) {
        this.currentFrame =
          frame

        this.renderer.renderFrame(
          frame
        )
      }

      this.animationFrameId =
        requestAnimationFrame(
          this.loop
        )
    } catch (error) {
      console.error(
        '鍔ㄧ敾寰幆寮傚父锛?,
        error
      )

      this.running = false

      if (
        typeof this.onError ===
        'function'
      ) {
        this.onError(error)
      }
    }
  }

  finishAnimation() {
    this.running = false

    const action =
      this.currentAction

    const {
      frameCount,
    } =
      action.spritesheet

    /*
     * 鍋滅暀鍦ㄦ渶鍚庝竴甯с€?
     */

    this.currentFrame =
      frameCount - 1

    this.renderer.renderFrame(
      this.currentFrame
    )

    if (
      typeof this.onAnimationEnd ===
      'function'
    ) {
      this.onAnimationEnd(
        action
      )
    }
  }
}
