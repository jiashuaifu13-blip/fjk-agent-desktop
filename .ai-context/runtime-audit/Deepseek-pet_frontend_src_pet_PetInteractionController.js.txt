import {
  PetStates,
} from './PetStates.js'


export class PetInteractionController {

  constructor({
    root,
    canvas,
    stateMachine,
    hitTest,
    dragThreshold = 6,
  }) {

    this.root =
      root

    this.canvas =
      canvas

    this.stateMachine =
      stateMachine

    this.hitTest =
      hitTest

    this.dragThreshold =
      dragThreshold


    this.pointerId =
      null

    this.pointerDown =
      false

    this.dragging =
      false


    this.startPointerX =
      0

    this.startPointerY =
      0

    this.startRootLeft =
      0

    this.startRootTop =
      0


    this.onInteraction =
      null


    this.handlePointerDown =
      this.handlePointerDown.bind(this)

    this.handlePointerMove =
      this.handlePointerMove.bind(this)

    this.handlePointerUp =
      this.handlePointerUp.bind(this)

    this.handlePointerCancel =
      this.handlePointerCancel.bind(this)

    this.handleResize =
      this.handleResize.bind(this)
  }


  mount() {

    this.canvas.addEventListener(
      'pointerdown',
      this.handlePointerDown
    )

    this.canvas.addEventListener(
      'pointermove',
      this.handlePointerMove
    )

    this.canvas.addEventListener(
      'pointerup',
      this.handlePointerUp
    )

    this.canvas.addEventListener(
      'pointercancel',
      this.handlePointerCancel
    )

    window.addEventListener(
      'resize',
      this.handleResize
    )
  }


  destroy() {

    this.canvas.removeEventListener(
      'pointerdown',
      this.handlePointerDown
    )

    this.canvas.removeEventListener(
      'pointermove',
      this.handlePointerMove
    )

    this.canvas.removeEventListener(
      'pointerup',
      this.handlePointerUp
    )

    this.canvas.removeEventListener(
      'pointercancel',
      this.handlePointerCancel
    )

    window.removeEventListener(
      'resize',
      this.handleResize
    )
  }


  async handlePointerDown(
    event
  ) {

    /*
     * 鍏抽敭锛?
     * 鍏堝垽鏂槸涓嶆槸瑙掕壊瀹炰綋鍍忕礌銆?
     */
    const hit =
      this.hitTest.hit(
        event.clientX,
        event.clientY
      )

    if (!hit) {

      if (
        typeof this.onInteraction ===
        'function'
      ) {
        this.onInteraction({
          type:
            'transparent-pointerdown',
        })
      }

      return
    }


    event.preventDefault()

    this.pointerDown =
      true

    this.dragging =
      false

    this.pointerId =
      event.pointerId


    this.startPointerX =
      event.clientX

    this.startPointerY =
      event.clientY


    /*
     * 鎶?right / bottom 甯冨眬
     * 杞崲鎴愭槑纭殑 left / top锛?
     * 鍚庣画鎷栧姩鏇寸畝鍗曘€?
     */
    const rect =
      this.root.getBoundingClientRect()

    this.startRootLeft =
      rect.left

    this.startRootTop =
      rect.top

    this.root.style.left =
      `${rect.left}px`

    this.root.style.top =
      `${rect.top}px`

    this.root.style.right =
      'auto'

    this.root.style.bottom =
      'auto'


    this.canvas.setPointerCapture(
      event.pointerId
    )


    if (
      typeof this.onInteraction ===
      'function'
    ) {
      this.onInteraction({
        type:
          'pointerdown',
      })
    }
  }


  async handlePointerMove(
    event
  ) {

    if (
      !this.pointerDown ||
      event.pointerId !==
      this.pointerId
    ) {
      return
    }


    const deltaX =
      event.clientX -
      this.startPointerX

    const deltaY =
      event.clientY -
      this.startPointerY


    const distance =
      Math.hypot(
        deltaX,
        deltaY
      )


    /*
     * 绉诲姩璺濈瓒呰繃闃堝€煎悗锛?
     * 鎵嶆寮忚涓虹敤鎴锋槸鍦ㄦ嫋鍔ㄣ€?
     *
     * 杩欐牱鏅€氱偣鍑讳笉浼氳鍒ゆ垚鎷栧姩銆?
     */
    if (
      !this.dragging &&
      distance >=
      this.dragThreshold
    ) {

      this.dragging =
        true

      try {
        await this.stateMachine
          .setState(
            PetStates.DRAGGING,
            {
              reason:
                'pointer-drag-start',
            }
          )
      } catch (error) {
        console.error(
          'Enter DRAGGING failed:',
          error
        )
      }
    }


    if (!this.dragging) {
      return
    }


    event.preventDefault()


    const proposedLeft =
      this.startRootLeft +
      deltaX

    const proposedTop =
      this.startRootTop +
      deltaY


    const position =
      this.clampPosition(
        proposedLeft,
        proposedTop
      )


    this.root.style.left =
      `${position.left}px`

    this.root.style.top =
      `${position.top}px`


    if (
      typeof this.onInteraction ===
      'function'
    ) {
      this.onInteraction({
        type:
          'drag',
        left:
          position.left,
        top:
          position.top,
      })
    }
  }


  async handlePointerUp(
    event
  ) {

    if (
      !this.pointerDown ||
      event.pointerId !==
      this.pointerId
    ) {
      return
    }


    event.preventDefault()


    try {
      if (
        this.canvas.hasPointerCapture(
          event.pointerId
        )
      ) {
        this.canvas.releasePointerCapture(
          event.pointerId
        )
      }
    } catch {
      /*
       * 鏌愪簺鎯呭喌涓?pointer capture
       * 宸茬粡鐢辨祻瑙堝櫒鑷姩閲婃斁銆?
       */
    }


    const wasDragging =
      this.dragging


    this.resetPointer()


    try {

      if (wasDragging) {

        /*
         * 鎷栧姩缁撴潫鍚庡洖鍒板緟鏈恒€?
         */
        await this.stateMachine.idle(
          'pointer-drag-end'
        )


        if (
          typeof this.onInteraction ===
          'function'
        ) {
          this.onInteraction({
            type:
              'drag-end',
          })
        }

        return
      }


      /*
       * 娌℃湁浜х敓鎷栧姩锛?
       * 杩欐槸涓€娆＄湡姝ｇ偣鍑汇€?
       */
      await this.stateMachine
        .setState(
          PetStates.INTERACT,
          {
            force: true,
            reason:
              'pointer-click',
          }
        )


      if (
        typeof this.onInteraction ===
        'function'
      ) {
        this.onInteraction({
          type:
            'click',
        })
      }

    } catch (error) {

      console.error(
        'Pointer interaction failed:',
        error
      )
    }
  }


  async handlePointerCancel(
    event
  ) {

    if (
      event.pointerId !==
      this.pointerId
    ) {
      return
    }


    const wasDragging =
      this.dragging


    this.resetPointer()


    if (wasDragging) {
      try {
        await this.stateMachine.idle(
          'pointer-cancel'
        )
      } catch (error) {
        console.error(
          error
        )
      }
    }
  }


  resetPointer() {

    this.pointerDown =
      false

    this.dragging =
      false

    this.pointerId =
      null
  }


  clampPosition(
    left,
    top
  ) {

    const rect =
      this.root.getBoundingClientRect()

    const maxLeft =
      Math.max(
        0,
        window.innerWidth -
        rect.width
      )

    const maxTop =
      Math.max(
        0,
        window.innerHeight -
        rect.height
      )


    return {
      left:
        Math.min(
          Math.max(
            0,
            left
          ),
          maxLeft
        ),

      top:
        Math.min(
          Math.max(
            0,
            top
          ),
          maxTop
        ),
    }
  }


  handleResize() {

    const rect =
      this.root.getBoundingClientRect()

    const position =
      this.clampPosition(
        rect.left,
        rect.top
      )

    this.root.style.left =
      `${position.left}px`

    this.root.style.top =
      `${position.top}px`

    this.root.style.right =
      'auto'

    this.root.style.bottom =
      'auto'
  }
}
